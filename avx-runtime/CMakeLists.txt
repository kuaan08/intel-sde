cmake_minimum_required(VERSION 3.16)
project(SDE_AVX_Lab_Runtime LANGUAGES CXX)

add_executable(avx_runtime src/avx_runtime.cpp)
target_compile_features(avx_runtime PRIVATE cxx_std_17)

if(MSVC)
  target_compile_options(avx_runtime PRIVATE /W4 /O2 /arch:SSE2)
else()
  target_compile_options(avx_runtime PRIVATE -Wall -Wextra -Wpedantic -O2 -fno-tree-vectorize -fno-lto)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    target_compile_options(avx_runtime PRIVATE -march=x86-64 -msse4.2 -mno-avx -mno-avx2 -mno-fma -mno-avx512f)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  set(SDE_AVX_TEST_RUNNER "" CACHE FILEPATH "Optional emulator executable, e.g. /usr/bin/qemu-x86_64")
  set(SDE_AVX_TEST_RUNNER_ARGS "" CACHE STRING "Optional space-separated arguments for the test runner")
  if(SDE_AVX_TEST_RUNNER)
    separate_arguments(SDE_AVX_TEST_RUNNER_ARGS_LIST NATIVE_COMMAND "${SDE_AVX_TEST_RUNNER_ARGS}")
    add_test(NAME independent_avx_runtime_self_test COMMAND ${SDE_AVX_TEST_RUNNER} ${SDE_AVX_TEST_RUNNER_ARGS_LIST} $<TARGET_FILE:avx_runtime> --self-test)
  else()
    add_test(NAME independent_avx_runtime_self_test COMMAND avx_runtime --self-test)
  endif()
endif()
